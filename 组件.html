    // =========================================================================
// 10-components/src/app/doctor-card/doctor-card.ts - 一个子组件
// -------------------------------------------------------------------------
// 它通过输入从父组件接收数据，输入用
// input() 声明：每个输入都是只读的 signal，它跟随
// 父组件传入的值。其类型在编译时检查：
// <app-doctor-card [doctor]="42"> 会被拒绝。
//
// 黄金法则：组件绝不修改它的输入。数据
// 从父组件流向子组件；若要请求更改，
// 子组件发出一个事件（参见示例 11）。
// =========================================================================

import { booleanAttribute, Component, computed, input } from '@angular/core';
import { Doctor } from '../doctor';

@Component({
  // 选择器：在父组件中表示此组件的标签
  selector: 'app-doctor-card',
  templateUrl: './doctor-card.html',
  styleUrl: './doctor-card.css',
})
export class DoctorCard {
  // input.required：父组件必须提供此输入
  readonly doctor = input.required<Doctor>();
  readonly rank = input.required<number>();
  // input(默认值)：可选的输入。booleanAttribute
  // 允许简单地写 <app-doctor-card compact> 表示 true
  readonly compact = input(false, { transform: booleanAttribute });

  // 基于输入的 computed：若父组件修改了输入则重新计算
  protected readonly initials = computed(() => `${this.doctor().firstName[0]}${this.doctor().lastName[0]}`);
}
    • 第 19 行：选择器：父组件将写[<app-doctor-card>]；
    • 第 25–29 行：组件的输入（[inputs]）——由其父组件传递给它的数据。[input()]创建一个只读信号，用于跟踪父组件传入的值：
      • 第 25–26 行：[input.required<T>()]：必需的输入。如果父组件的模板中省略了[[doctor]]，将导致编译错误；
      • 第 29 行：[input(default value)]：可选的输入。[transform]选项用于转换接收到的值：[booleanAttribute]将属性的存在（[<app-doctor-card compact>]，没有值）转换为[true]，就像 HTML 属性[disabled]一样；
    • 第 32 行：一个基于输入的 [computed]：如果父组件改变了输入，就会重新计算。

    输入不是[protected]的：它们是组件公共接口的一部分，也就是其父组件所使用的接口。

    黄金法则：组件绝不修改自己的输入（无论如何它们都是只读的）。数据从父组件流向子组件；若要请求更改，子组件会发出事件（示例 11）。

// -------------------------------------------------------------------------
// 它导入 DoctorCard 组件（imports 数组），并将其作为
// 一个新的 HTML 标签 <app-doctor-card> 使用，可按
// 需使用任意多次：组件是界面中可复用的部分，拥有

import { Component, signal } from '@angular/core';
import { FormsModule } from '@angular/forms';

import { Doctor } from './doctor';

  imports: [FormsModule, DoctorCard],
  templateUrl: './app.html',
})
export class App {
  protected readonly doctors = signal<Doctor[]>([

    // =========================================================================
// 11-outputs-model/src/app/counter/counter.ts
// -------------------------------------------------------------------------
// 通知父组件的子组件：output() 声明组件
// 可以发出的事件，并带有其数据的类型。父组件像监听
// HTML 事件一样监听它：<app-counter (incremented)="..." (limit)="...">。
// =========================================================================

import { Component, input, output, signal } from '@angular/core';

@Component({
  selector: 'app-counter',
  template: `
    <div class="d-flex align-items-center gap-2 mb-2">
      <span class="w-25">{{ label() }}</span>
      <button class="btn btn-sm btn-outline-primary" (click)="plus()">+1</button>
      <span class="badge text-bg-secondary">{{ value() }} / {{ max() }}</span>
    </div>
  `,
})
export class Counter {
  readonly label = input.required<string>();
  readonly max = input.required<number>();

  // 两个事件：incremented（带新值）和 limit（无数据）
  readonly incremented = output<number>();
  readonly limit = output<void>();

  protected readonly value = signal(0);

  protected plus(): void {
    if (this.value() >= this.max()) {
      this.limit.emit();
      return;
    }
    this.value.update((v) => v + 1);
    this.incremented.emit(this.value());
  }
}
    • 第 13–19 行：组件的模板，写在装饰器中（属性 [template]，用反引号括起来）；
    • 第 26–27 行：[output<T>()] 声明一个输出：组件可以发出的事件，以及其数据的类型。[limit] 没有数据（[void]）；
    • 第 33 行和第 37 行：[emit(data)] 发出事件；父组件接收数据。

    避免以浏览器事件命名输出（[change]、[click]……）：组件内部元素发出的浏览器事件会向上冒泡到其宿主元素，父组件也会接收到它们。

    // =========================================================================
// 11-outputs-model/src/app/text-field/text-field.ts
// -------------------------------------------------------------------------
// 一个"自制"的表单字段（标签 + 字段 + 帮助 + 字符
// 计数器），像 ngModel 一样支持双向绑定：
//     <app-text-field [(value)]="lastName" label="Last name" [max]="20" />
//
// model() 创建一个与父组件关联的可写 signal：读取它得到
// 父组件的值，写入它（set）会发出 valueChange 事件，
// [( )] 语法再把该事件转为对父组件 signal 的更新。数据
// 因此真正仍归父组件所有。
// =========================================================================

import { Component, input, model } from '@angular/core';

// 由所有实例共享的计数器：每个字段一个唯一标识符，
// 用于关联 <label for> 和 <input id>
let nextNumber = 0;

@Component({
  selector: 'app-text-field',
  template: `
    <div class="mb-3">
      <label class="form-label" [for]="id">{{ label() }}</label>
      <!-- [value]：父组件的值；(input)：每次按键都会替换它 -->
      <input class="form-control" [id]="id" [value]="value()" [attr.maxlength]="max()"
             [class.is-invalid]="value().length === max()" (input)="onInput($event)">
      <div class="form-text">{{ help() }} ({{ value().length }} / {{ max() }})</div>
    </div>
  `,
})
export class TextField {
  readonly value = model.required<string>();
  readonly label = input.required<string>();
  readonly max = input.required<number>();
  readonly help = input('');

  protected readonly id = `text-field-${++nextNumber}`;

  protected onInput(event: Event): void {
    this.value.set((event.target as HTMLInputElement).value);
  }
}
    • 第 33 行：[model()] 创建一个可写信号，并与父组件关联。读取它会返回父组件的值；写入它（[set]，第 41 行）会修改该值并发出事件 [valueChange]（模型名称后加 [Change]）。因此，使用 [[(value)]="signal"] 进行写入的父组件，其信号会被更新；
    • 第 18 行：一个模块变量（位于类之外）由组件的所有实例共享：一个为每个字段分配唯一标识符的计数器（第 38 行），用于关联 [<label for>] 和 [<input id>]；
    • 第 26–27 行：[[value]] 显示父组件的值；[(input)] 在每次按键时替换它（第 41 行）。[[attr.maxlength]] 是对 HTML 的 attribute 绑定（attribute [maxlength]），而不是 property 绑定。

    // =========================================================================
// 11-outputs-model/src/app/app.ts - 父组件监听其子组件
// -------------------------------------------------------------------------
// 子组件向上传递信息的两种方式：
//   1. 发出事件（output），父组件用 ( ) 监听；
//   2. 在 model() 上使用双向绑定 [( )]：它是输入 [value]
//      + 事件 (valueChange) 的简写。
// =========================================================================

import { Component, computed, signal } from '@angular/core';
import { Counter } from './counter/counter';
import { TextField } from './text-field/text-field';

@Component({
  selector: 'app-root',
  imports: [Counter, TextField],
  templateUrl: './app.html',
})
export class App {
  protected readonly total = signal(0);
  protected readonly message = signal('');
  protected readonly firstName = signal('Marie');
  protected readonly lastName = signal('PELISSIER');
  protected readonly fullName = computed(() => `${this.firstName()} ${this.lastName().toUpperCase()}`);

  // incremented 事件的处理函数：它接收所发出的数据（$event）
  protected onChange(label: string, value: number): void {
    this.total.update((t) => t + 1);
    this.message.set(`${label} is now ${value}`);
  }

  protected clear(): void {
    this.firstName.set('');
    this.lastName.set('');
  }
}

// 12-projection/src/app/card/card.ts - 一个可复用的模板
// -------------------------------------------------------------------------
// 输入传递数据；内容投影传递
// 内容（HTML、其他组件），由父组件写在
// <app-card> 和 </app-card> 标签之间。卡片确定外框（边框，
// 页眉、页脚），父组件提供其中的内容：

//   - 放在 <ng-content> 和 </ng-content> 之间的内容，在

// 在客户端的对应物。

@Component({
  selector: 'app-card',
  template: `
    <div class="card mb-3" [class]="color() ? 'border-' + color() : ''">
      <div class="card-header" [class]="color() ? 'text-bg-' + color() : ''">
        <ng-content select="[header]">Untitled</ng-content>
      </div>
      <div class="card-body">

// 13-lifecycle/src/app/clock/clock.ts
// -------------------------------------------------------------------------
// 组件的生命周期：
//   1. constructor ：创建类；输入此时
//                         尚不可用；
//   2. ngOnInit ：输入已接收到第一个值；
//   3. afterNextRender ：组件已显示，其 HTML 元素
//                         已存在于页面中；
//   4. ngOnDestroy ：组件即将从页面中移除。
// 时钟在启动时开启一个定时器（setInterval）……并且必须
// 在它消失时务必停止该定时器：否则定时器
// 会白白一直运行下去（内存泄漏）。

import { afterNextRender, Component, OnDestroy, OnInit, output, signal } from '@angular/core';

@Component({
  selector: 'app-clock',
  template: `<div class="display-6 font-monospace">{{ time() }}</div>`,
})
// implements OnInit, OnDestroy：TypeScript 检查这些方法是否存在
export class Clock implements OnInit, OnDestroy {
  readonly log = output<string>();
  protected readonly time = signal(new Date().toLocaleTimeString());
  private timer: number | undefined;

  constructor() {
    console.log('Clock: constructor'); // 父组件尚未监听 log
    // afterNextRender 在构造函数中注册；该函数
    // 在组件首次显示之后立即调用一次
    afterNextRender(() => this.log.emit('Clock: afterNextRender - displayed'));
  }

  ngOnInit(): void {
    this.log.emit('Clock: ngOnInit - starting the timer');
    this.timer = window.setInterval(() => this.time.set(new Date().toLocaleTimeString()), 1000);
  }

  ngOnDestroy(): void {
    window.clearInterval(this.timer);
    this.log.emit('Clock: ngOnDestroy - timer stopped');
  }
}

    步骤：

    • 第 28–33 行：构造函数：类被创建。输入尚未被赋值，父组件也尚未监听输出（因此第 29 行使用 [console.log] ，而不是 [emit]）；
    • 第 35–38 行：[ngOnInit()]：输入已收到其第一个值。这是触发依赖它们的操作的常用位置：加载、定时器；
    • 第 32 行：[afterNextRender(function)]，在构造函数中调用，注册一个只执行一次的函数，在组件被首次显示后立即执行：其 HTML 元素此时已存在于页面上；
    • 第 40–43 行：[ngOnDestroy()]：组件即将从页面移除。定时器被停止。

    [implements OnInit, OnDestroy] （第 23 行）不是必需的：[Angular] 会调用 [ngOnInit] （如果存在）。但 [TypeScript] 随后会验证这些方法是否书写正确。

    [Angular] 提供了其他阶段，自从引入信号以来，它们的用处变小了：[ngOnChanges]（某个输入发生了变化；对于该输入，我们更倾向于使用 [computed] 或 [effect] ）、[ngAfterViewInit]……以及 [ngOnDestroy]的一种替代方式：[inject(DestroyRef).onDestroy(function)]，示例 14 将使用它。

    3.5. 示例 14：服务与依赖注入

    一个服务 是将多个组件共享的数据和代码组合在一起的类：服务器访问、已登录用户、购物车等。组件不会创建服务（绝不要 [new CartService()]）：它们会请求这些服务，请求对象是 [Angular]，方式是使用 [inject()] 函数。这就是 依赖注入，即 [NestJS] 课程在服务器端介绍过的机制。

import { DestroyRef, Directive, inject, input, output, signal } from '@angular/core';

@Directive({
  selector: '[appLongPress]',
  host: {
    '[class.pressed]': 'pressing()',
    '(pointerdown)': 'start()',
    '(pointerup)': 'stop()',
    '(pointerleave)': 'stop()',
  },
})
export class LongPressDirective {
  readonly duration = input(1000); // 以毫秒为单位
  readonly appLongPress = output<void>();
  protected readonly pressing = signal(false);
  private timer: number | undefined;

  constructor() {
    inject(DestroyRef).onDestroy(() => window.clearTimeout(this.timer));
  }

  protected start(): void {
    this.pressing.set(true);
    this.timer = window.setTimeout(() => {
      this.pressing.set(false);
      this.appLongPress.emit();
    }, this.duration());
  }

  protected stop(): void {
    window.clearTimeout(this.timer);
    this.pressing.set(false);
  }
}

    // =========================================================================
// 16-confirmation/src/app/confirmation/confirmation.service.ts
// -------------------------------------------------------------------------
// "Are you sure?"：window.confirm() 虽然存在，但既不能
// 设置样式，其按钮也无法翻译。我们用由本服务驱动的
// Bootstrap 对话框（confirm-dialog.ts）取而代之。
//
// 诀窍：confirm(message) 返回一个 PROMISE，用户作出回答时
// 解析为 true（是）或 false（否）。调用代码
// 因此可以写得和 window.confirm 一样简单：
//
//     if (await this.confirmation.confirm('Delete this doctor?')) { ... }
//
// providedIn: 'root'：整个应用只有一个服务，因此也只有一个
// 对话框，在根组件中只放置一次。
// =========================================================================

import { Injectable, signal } from '@angular/core';

@Injectable({ providedIn: 'root' })
export class ConfirmationService {
  private readonly _message = signal<string | null>(null); // null：对话框关闭
  readonly message = this._message.asReadonly();
  private answer: ((yes: boolean) => void) | null = null;

  confirm(text: string): Promise<boolean> {
    this.answer?.(false); // 还有未回答的问题？视为“否”
    this._message.set(text);
    return new Promise<boolean>((resolve) => {
      this.answer = resolve;
    });
  }

  respond(yes: boolean): void {
    this._message.set(null);
    const r = this.answer;
    this.answer = null;
    r?.(yes);
  }
}
    • 第 20 行：整个应用程序只有一个服务；因此只会有一个窗口，一次性放置在根组件中；
    • 第 22 行：显示的消息；[null] 表示窗口已关闭；
    • 第 23 行：[asReadonly()]：组件可以读取消息，但只有服务才能修改它；
    • 第 24 行：用于解决当前 promise 的函数，该 promise 正在等待响应；
    • 第 26–32 行：[confirm] 打开窗口（第 28 行）并返回一个 promise。它的 [resolve] 函数被保存起来（第 30 行）：直到稍后用户点击按钮时，该 promise 才会被解决；
    • 第 27 行：如果还有未关闭的问题，它会被回答为“否”：其 promise 不会永远处于挂起状态；
    • 第 34–39 行：[respond] 由窗口调用：它关闭窗口（第 35 行），然后用用户的响应解决 promise（第 38 行）。

    这就是 promise 的好处：提出问题的代码把它写成对 [window.confirm()] 的调用，等待响应时使用 [await]，无需回调函数。

  imports: [ConfirmDialog],
  templateUrl: './app.html',
})
export class App {
  private readonly confirmation = inject(ConfirmationService);
  protected readonly contacts = signal<Contact[]>([
    { id: 1, name: 'Jean MARTIN' },
    { id: 2, name: 'Alice DURAND' },
    { id: 3, name: 'Luc GERMAN' },
  ]);
  protected readonly message = signal('');

  // 异步方法：它“等待”用户的回答
  protected async remove(c: Contact): Promise<void> {
    if (!(await this.confirmation.confirm(`Delete the contact ${c.name}?`))) {
      this.message.set('Deletion cancelled.');
      return;
    }
    this.contacts.update((list) => list.filter((x) => x.id !== c.id));
    this.message.set(`${c.name} has been deleted.`);
  }

  protected async removeAll(): Promise<void> {
    if (await this.confirmation.confirm(`Delete the ${this.contacts().length} contacts?`)) {
      this.contacts.set([]);
      this.message.set('All the contacts have been deleted.');
    }
  }
}
