    // =========================================================================
// 01-first-application/src/app/app.ts - 第一个组件
// -------------------------------------------------------------------------
// Angular 组件是一个由 @Component 装饰的 TypeScript 类：
//   - selector ：表示它的 HTML 标签（<app-root>）；
//   - templateUrl ：其 HTML 文件（app.html）；
//   - styleUrl ：其 CSS 文件（app.css），仅作用于
//                   此组件。
// 类的属性和方法可以在模板中使用。
// =========================================================================

import { Component, signal } from '@angular/core';

@Component({
  selector: 'app-root',
  templateUrl: './app.html',
  styleUrl: './app.css',
})
export class App {
  // "普通"数据：模板可以显示它
  protected readonly title = 'Welcome to Angular';
  protected readonly languages = ['TypeScript', 'HTML', 'CSS'];

  // signal(...)：一个响应式数据。通过调用它来读取其值
  // （counter()），用 set() 或 update() 修改。Angular 随后会
  // 自动更新页面中显示它的位置。
  protected readonly counter = signal(0);
  protected readonly time = signal(new Date().toLocaleTimeString());

  // 由模板中的按钮调用的方法
  protected increment(): void {
    this.counter.update((n) => n + 1);
    this.time.set(new Date().toLocaleTimeString());
  }
}
    • 第 14–18 行：[@Component] 装饰器将该类变成组件：
      • 第 15 行：它的 选择器，即在模板或 [index.html] 中代表它的标签；
      • 第 16 行：它的模板文件，HTML；
      • 第 17 行：它的样式文件，CSS；
    • 第 21–22 行：两个普通数据项。[protected]：它们仅由组件及其模板使用；[readonly]：它们不会被替换；
    • 第 27 行：[signal(0)] 创建一个响应式数据点，初始值为 0。信号是一个函数：通过调用它（[this.counter()]）来读取其值，并通过 [set(value)] 或 [update(function)]来修改它。当它变化时，[Angular] 会更新页面上所有显示它的位置；
    • 第 31–34 行：一个修改两个信号的方法。它不会改动页面：[Angular] 会更新页面。[update((n) => n + 1)] 根据旧值计算新值。

    // =========================================================================
// 02-signals/src/app/app.ts - signal、computed、effect、linkedSignal
// -------------------------------------------------------------------------
// 一个含增值税价格的小型计算器：
//   - signal ：响应式的值，通过调用来读取，
//                    通过 set() / update() 修改；
//   - computed ：由其他 signal 计算得出的值，
//                    当它们变化时自动重新计算；
//   - effect ：在其读取的 signal 每次变化时
//                    执行的代码（"副作用"）；
//   - linkedSignal ：可写的 signal，但当另一个 signal 变化时，
//                    会被重置为一个计算得出的值。
// =========================================================================

import { Component, computed, effect, linkedSignal, signal, untracked } from '@angular/core';
// FormsModule 提供 [(ngModel)]，用于将字段与 signal 绑定（参见示例 05）
import { FormsModule } from '@angular/forms';

@Component({
  selector: 'app-root',
  imports: [FormsModule],
  templateUrl: './app.html',
})
export class App {
  protected readonly label = signal('Pen');
  protected readonly priceExclVat = signal(2.5);
  protected readonly vatRate = signal(20); // 单位为 %

  // linkedSignal：每当商品改变时数量恢复为 1，
  // 但在此期间用户可以修改它
  protected readonly quantity = linkedSignal({ source: this.label, computation: () => 1 });

  // computed：从不直接赋值，总是重新计算。Angular 知道
  // 它依赖哪些 signal（即它所读取的那些）：当其中任何一个
  // 变化时才重新计算，且仅在此时重新计算
  protected readonly totalExclVat = computed(() => this.quantity() * this.priceExclVat());
  protected readonly vatAmount = computed(() => (this.totalExclVat() * this.vatRate()) / 100);
  protected readonly totalInclVat = computed(() => this.totalExclVat() + this.vatAmount());

  // 欧元金额格式化器，英语格式（€1,234.50）
  private readonly euros = new Intl.NumberFormat('en-US', { style: 'currency', currency: 'EUR' });

  // 含增值税总额的历史记录，由下面的 effect 填充
  protected readonly history = signal<string[]>([]);
  private oldTotal: number | null = null;

  constructor() {

    effect(() => {

        // untracked：修改 history 而不使其成为该 effect 的依赖
        untracked(() => this.history.update((h) => [line, ...h].slice(0, 5)));
      }
      this.oldTotal = total;
    });

    return this.euros.format(amount);

    this.label.set('Pen');
    this.priceExclVat.set(2.5);
    this.vatRate.set(20);

// -------------------------------------------------------------------------
// (event)="statement"：当元素发出该事件时
// 执行此语句：
//   - (click)="f()" 调用 f；
//   - (click)="f(3, $event)" 带参数调用 f；$event 是
//                              浏览器的原生事件；
//   - (keyup.enter)、(keydown.escape)：仅针对该按键；
//   - 元素上的 #name：模板的"引用变量"，它
//     指向 HTML 元素本身（name.value、name.focus()...）。

import { Component, signal } from '@angular/core';

interface Task {
  id: number;
  label: string;
  done: boolean;
}

@Component({
  selector: 'app-root',
  templateUrl: './app.html',
  styles: `
    .area {
      background: #e9f2ff;
      height: 80px;
      cursor: crosshair;
    }
  `,
})
export class App {
  protected readonly tasks = signal<Task[]>([
    { id: 1, label: 'Install Node.js', done: true },
    { id: 2, label: 'Read the chapter on events', done: false },
  ]);
  protected readonly log = signal<string[]>([]);
  private nextId = 3;

  private trace(message: string): void {
    this.log.update((j) => [`${new Date().toLocaleTimeString()} ${message}`, ...j]);
  }

  // 由 (submit) 调用：$event.preventDefault() 阻止浏览器
  // 将表单发送到服务器（以及重新加载页面）
  protected add(event: SubmitEvent, field: HTMLInputElement): void {
    event.preventDefault();
    const label = field.value.trim();
    if (label === '') {
      return;
    }
    // 包含数组的 signal：我们用一个新数组来替换它
    this.tasks.update((t) => [...t, { id: this.nextId++, label, done: false }]);
    this.trace(`added "${label}"`);
    field.value = '';
    field.focus();
  }

  protected toggle(task: Task, done: boolean): void {
    this.tasks.update((t) => t.map((x) => (x.id === task.id ? { ...x, done } : x)));
    this.trace(`"${task.label}": ${done ? 'done' : 'to do'}`);
  }

  protected remove(task: Task): void {
    this.tasks.update((t) => t.filter((x) => x.id !== task.id));
    this.trace(`deleted "${task.label}"`);
  }

  // $event：原生事件（此处为 MouseEvent），带有其坐标
  protected position(event: MouseEvent): void {
    this.trace(`click at (${event.offsetX}, ${event.offsetY})`);
  }
}

  lastName: string;
  age: number;
  newsletter: boolean;
  email: string;
  country: string;
  birthDate: string;
  presentation: string;

// 07-validation/src/app/app.ts - 在浏览器中验证表单
// -------------------------------------------------------------------------
// 字段的规则写在表单的 SCHEMA 中（form() 的第二个
// 参数）：required、maxLength、minLength、pattern... 以及
// 用于"自制"规则的 validate()。随后每个字段都知道自己的
// 错误（f.lastName().errors()），每次按键都会重新计算。
// 这些错误不会立即显示（满是红色的空表单
// 令人不快）：只对已离开（"touched"）的字段显示。submit()
// 将所有字段标记为已触碰，然后只有在
// 表单有效时才执行操作。
//
// 与 RdvMedecins 服务器的规则相同（"Create an account" 表单）。
// 警告：这种验证只是为用户提供的便利（即时
// 响应，无需往返服务器）。它不保护
// 任何东西：任何人都可以向服务器发送手工
// 构造的请求（curl...）。因此服务器始终会重新验证一切（参见示例 25）。
// =========================================================================

import { Component, signal } from '@angular/core';
import { form, FormField, maxLength, minLength, pattern, required, submit, validate } from '@angular/forms/signals';

interface Account {
  title: string;
  firstName: string;
  lastName: string;
  login: string;
  password: string;
  confirmation: string;
}

const TITLES = ['Mr', 'Mme', 'Melle'];
const LOGIN_FORMAT = /^[a-z0-9._-]{1,30}$/;
const PASSWORD_MIN = 6;

@Component({
  selector: 'app-root',
  imports: [FormField],
  templateUrl: './app.html',
})
export class App {
  protected readonly titles = TITLES;
  protected readonly model = signal<Account>({
    title: 'Mr',
    firstName: '',
    lastName: '',
    login: '',
    password: '',
    confirmation: '',
  });

  // schema：p 指代各字段的路径（p.lastName、p.login...）
  protected readonly f = form(this.model, (p) => {
    validate(p.title, ({ value }) =>
      TITLES.includes(value()) ? null : { kind: 'title', message: 'Choose a title from the list.' },
    );
    required(p.firstName, { message: 'The first name is required.' });
    maxLength(p.firstName, 20, { message: 'The first name must not exceed 20 characters.' });
    required(p.lastName, { message: 'The last name is required.' });
    maxLength(p.lastName, 20, { message: 'The last name must not exceed 20 characters.' });
    required(p.login, { message: 'The login is required.' });
    pattern(p.login, LOGIN_FORMAT, { message: 'Unaccented lower-case letters, digits, dot, hyphen, underscore.' });
    required(p.password, { message: 'The password is required.' });
    minLength(p.password, PASSWORD_MIN, { message: `At least ${PASSWORD_MIN} characters.` });
    // 一条"自制"规则，比较两个字段（valueOf 读取另一个字段）
    validate(p.confirmation, ({ value, valueOf }) =>
      value() === valueOf(p.password)
        ? null
        : { kind: 'confirmation', message: 'The confirmation does not match the password.' },
    );
  });

  protected readonly submitAttempted = signal(false);
  protected readonly success = signal(false);

  // 表单的 (submit)
  protected async send(event: SubmitEvent): Promise<void> {
    event.preventDefault();
    this.submitAttempted.set(true);
    // submit() 将所有字段标记为已触碰（其错误
    // 随之出现），然后只有在全部有效时才执行操作
    const valid = await submit(this.f, async () => {
      // 此处我们会把模型发送到服务器（参见示例 25）
      return null; // null：无错误
    });
    this.success.set(valid);
  }
}

    • 第 53–71 行：表单模式：每个函数将一条规则与一个字段（由其路径标识，[p.firstName]）关联起来，同时附带验证失败时显示的消息：
      • [required]：字段必须填写；
      • [maxLength]、[minLength]：最大或最小长度；
      • [pattern]：值必须匹配某个正则表达式；
      • [validate(path, function)]：“自定义”规则。该函数接收字段的上下文（它的值[value()]、另一个字段的值[valueOf(...)]），并返回[null]（无错误）或一个错误：一个对象[{ kind, message }]，其中[kind]表示错误类型（第 54–56 行和第 66–70 行）。确认规则比较两个字段：当其中任何一个变化时都会重新计算；

// 08-reactive-forms/src/app/app.ts - "历史"表单 API
// -------------------------------------------------------------------------
// 在信号表单（Angular 22）之前，表单是用
// 响应式表单（ReactiveFormsModule）编写的。它们存在于
// 大多数现有应用中：这里是示例
// 07 的表单，用它们重写。
//   - FormGroup 汇集多个 FormControl，每个字段一个；
//   - 每个 FormControl 有一个初始值和验证器
//     （Validators.required、Validators.maxLength(20)...）；
//   - 在模板中，[formGroup] 将 <form> 与该组绑定，而
//     formControlName="lastName" 将 HTML 字段与一个控件绑定；
//   - 控件的状态（value、errors、touched、invalid...）是
//     普通属性，而不是 signal。
// =========================================================================

import { Component, inject, signal } from '@angular/core';
import { AbstractControl, FormBuilder, ReactiveFormsModule, ValidationErrors, Validators } from '@angular/forms';

const TITLES = ['Mr', 'Mme', 'Melle'];

// "自制"验证器作用于整个组：它比较组中的两个字段。
// 它返回 null（无错误）或描述该错误的对象。
function sameConfirmation(group: AbstractControl): ValidationErrors | null {
  const password = group.get('password')?.value;
  const confirmation = group.get('confirmation')?.value;
  return password === confirmation ? null : { confirmation: true };
}

@Component({
  selector: 'app-root',
  imports: [ReactiveFormsModule],
  templateUrl: './app.html',
})
export class App {
  protected readonly titles = TITLES;
  private readonly fb = inject(FormBuilder).nonNullable;

  // 组：{ name: [初始值, 验证器] }
  protected readonly form = this.fb.group(
    {
      title: ['Mr', [Validators.required]],
      firstName: ['', [Validators.required, Validators.maxLength(20)]],
      lastName: ['', [Validators.required, Validators.maxLength(20)]],
      login: ['', [Validators.required, Validators.pattern(/^[a-z0-9._-]{1,30}$/)]],
      password: ['', [Validators.required, Validators.minLength(6)]],
      confirmation: [''],
    },
    { validators: sameConfirmation },
  );

  protected readonly success = signal(false);
  protected readonly value = signal('');

  // 如果字段已被触碰且无效，则标记为出错
  protected inError(name: string): boolean {
    const control = this.form.get(name);
    return !!control && control.touched && control.invalid;
  }

  // (ngSubmit)：所有字段变为"已触碰"，其错误随之出现
  protected send(): void {
    this.form.markAllAsTouched();
    this.success.set(this.form.valid);
    // getRawValue()：整个组的值，带类型
    this.value.set(JSON.stringify(this.form.getRawValue()));
  }
}
    • 第 18 行：响应式表单的类，位于 [@angular/forms]；

    // =========================================================================
// 09-pipes/src/app/pipes/time-range.pipe.ts - 一个"自制"管道
// -------------------------------------------------------------------------
// 管道是一个由 @Pipe 装饰的类，其 transform() 方法
// 接收放在竖线之前的值（以及可能的
// 参数），并返回要显示的内容：
//     {{ slot | timeRange }} -> "8:00 - 8:20"
//     {{ slot | timeRange: 'start' }} -> "8:00"
// 管道默认是"纯"的：只有当值或参数发生变化时，Angular 才会
// 再次调用 transform()（同一个对象 = 同一个结果）。
// =========================================================================

import { Pipe, PipeTransform } from '@angular/core';

export interface Slot {
  startHour: number;
  startMinute: number;
  endHour: number;
  endMinute: number;
}

// "8:05" 而不是 "8:5"
function time(h: number, m: number): string {
  return `${h}:${String(m).padStart(2, '0')}`;
}

@Pipe({ name: 'timeRange' })
export class TimeRangePipe implements PipeTransform {
  transform(c: Slot, part: 'all' | 'start' | 'end' = 'all'): string {
    switch (part) {
      case 'start':
        return time(c.startHour, c.startMinute);
      case 'end':
        return time(c.endHour, c.endMinute);
      default:
        return `${time(c.startHour, c.startMinute)} - ${time(c.endHour, c.endMinute)}`;
    }
  }
}
    • 第 27 行：管道的名称，即在模板中书写的名称；
    • 第 29 行：[transform(value, part)]返回应该显示的内容。参数是可选的（默认值['all']）：[{{ slot | timeRange }}]显示[8:00 - 8:20]，[{{ slot | timeRange: 'start' }}]显示[8:00]。

    管道默认是纯（pure）的：[Angular]只会调用[transform]，前提是值或其中某个参数发生了变化（不同的对象、不同的字符串）。这使它比在模板中调用组件方法（例如示例 02 中的[format(...)]）更高效，后者在每次检查组件时都会执行。

export class InitialsPipe implements PipeTransform {
  transform(fullName: string): string {
    return fullName
      .split(/\s+/)
      .filter((word) => word !== '')
      .map((word) => word[0].toUpperCase())
